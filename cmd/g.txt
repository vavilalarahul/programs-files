# 1. Check Git configuration
git config --global --list

# 2. Change username globally
git config --global user.name "Your Name"

# 3. Change email globally
git config --global user.email "your-email@example.com"

# 4. Change username for current repository
git config user.name "Your Name"

# 5. Change email for current repository
git config user.email "your-email@example.com"

# 6. Clone repository
git clone <repository-url>

# 7. Initialize repository
git init

# 8. Check status
git status

# 9. Add all files
git add .

# 10. Commit changes
git commit -m "message"

# 11. View commit history
git log --oneline

# 12. View changes
git diff

# 13. View staged changes
git diff --staged

# 14. View remote repositories
git remote -v

# 15. Pull changes
git pull origin main

# 16. Push changes
git push origin main

# 17. List branches
git branch

# 18. Switch branch
git checkout branch-name

# 19. Create and switch branch
git checkout -b new-branch

# 20. Switch to main branch
git checkout main

# 21. Merge branch
git merge branch-name

# 22. Restore file changes
git restore file.txt

# 23. Revert a commit
git revert <commit-id>

# 24. Undo commit but keep changes staged
git reset --soft HEAD~1

# 25. Undo commit and discard changes
git reset --hard HEAD~1

# 26. Remove file
git rm file.txt

# 27. Rename or move file
git mv old-name new-name

# 28. Save changes temporarily
git stash

# 29. Restore stashed changes
git stash pop

# 30. Delete local branch
git branch -d branch-name
