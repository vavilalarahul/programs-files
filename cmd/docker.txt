B2) BASIC COMPOSE YAML (USE THIS FOR COMPOSE COMMAND PRACTICE)
Create a folder, then save this as compose.yaml inside it:

services:
  web:
    image: nginx:latest
    ports:
      - "8080:80"

Folder: compose-lab/compose.yaml
Run these from the folder containing compose.yaml:
docker compose up -d
docker compose ps
docker compose logs
docker compose exec web sh
docker compose stop
docker compose start
docker compose restart
docker compose down
Open: http://localhost:8080

D) COMPOSE YAML (NGINX + POSTGRES + REDIS + NETWORK + VOLUME)
Create folder compose-lab and save as compose.yaml:

services:
  web:
    image: nginx:latest
    ports:
      - "8080:80"
    depends_on:
      - db
      - redis
    networks:
      - app-net

  db:
    image: postgres:15
    environment:
      POSTGRES_USER: demo
      POSTGRES_PASSWORD: demo
      POSTGRES_DB: demo_db
    volumes:
      - db-data:/var/lib/postgresql/data
    networks:
      - app-net

  redis:
    image: redis:alpine
    networks:
      - app-net

networks:
  app-net:

volumes:
  db-data:

Run/check/stop:
docker compose up -d
docker compose ps
docker compose logs
docker compose down
docker compose up -d
docker compose down -v   (also deletes db-data)
Open: http://localhost:8080

E) WORDPRESS + MYSQL YAML (ALTERNATE EXERCISE)
Save as docker-compose.yml:

services:
  wordpress:
    image: wordpress:latest
    ports:
      - "8081:80"
    environment:
      WORDPRESS_DB_HOST: db:3306
      WORDPRESS_DB_USER: wordpress
      WORDPRESS_DB_PASSWORD: wordpress
      WORDPRESS_DB_NAME: wordpress
    depends_on:
      - db

  db:
    image: mysql:5.7
    environment:
      MYSQL_ROOT_PASSWORD: rootpassword
      MYSQL_DATABASE: wordpress
      MYSQL_USER: wordpress
      MYSQL_PASSWORD: wordpress

Run: docker compose up -d
Open: http://localhost:8081
Stop: docker compose down

EXAM FLOW
Dockerfile: create folder > Dockerfile > docker build -t redisnew . >
docker run --name myredisnew -d redisnew > docker ps > stop > rm.
Compose: create folder > compose.yaml > docker compose up -d >
docker compose ps > localhost:8080 > docker compose down.

-d = background mode; -p HOST:CONTAINER; depends_on controls startup order,
not readiness. Demo credentials are for lab use only.
